T1¶

In [12]:
plt.figure(figsize=(12, 4))

for i, type in enumerate(cell_types):

    # T1 fitting:
    data = df[df['Cell_type'] == type]['T1'].apply(pd.to_numeric, errors='coerce').dropna()
    div = 'T1'

    plt.subplot(1, 3, i + 1)

    plt.hist(data, bins=60, density=True, color='skyblue', edgecolor='black', linewidth=0.1)

    shape, _, scale = lognorm.fit(data, floc=0)
    mu = np.log(scale) # ln
    sigma = shape
    
    mu_list.append(mu)
    mu_list_T1.append(mu)
    sigma_list.append(sigma)

    x = np.linspace(0, data.max()+10, 500)
    pdf = lognorm.pdf(x, shape, loc=0, scale=scale)
    plt.plot(x, pdf, 'r-', label=f'μ={mu:.2f}, σ={sigma:.2f}')

    plt.title(f'{type}')
    plt.xlabel('Time (hours)')
    plt.ylabel('Density')
    plt.legend()
    plt.grid(linestyle='--')
    # fix axis if needed:
    plt.xlim(0, 100)
    plt.ylim(0, 0.225)

plt.suptitle(f'{exp} - T1')
plt.tight_layout()
plt.show()
